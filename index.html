async function loadData() { const year = document.getElementById("yearSelect").value; const tbody = document.querySelector("#weatherTable tbody"); tbody.innerHTML = "Caricamento in corso..."; // URL con parametri corretti e controllo date const url = `https://archive-api.open-meteo.com/v1/archive?latitude=45.0483&longitude=9.6953&start_date=${year}-01-01&end_date=${year}-12-31&daily=temperature_2m_max,temperature_2m_min,apparent_temperature_max,relative_humidity_2m_mean,pressure_msl_mean,weather_code,precipitation_sum,snowfall_sum,wind_speed_10m_max,wind_direction_10m_dominant&timezone=auto`; try { const response = await fetch(url); if (!response.ok) throw new Error('Network response was not ok'); const data = await response.json(); // Se i dati daily non esistono (possibile in anni molto vecchi) if (!data.daily) { tbody.innerHTML = "Dati non disponibili per questo anno."; return; } const d = data.daily; let htmlRows = ""; for (let i = d.time.length - 1; i >= 0; i--) { let precipDisplay = (d.precipitation_sum[i] != null && d.precipitation_sum[i] > 0) ? d.precipitation_sum[i] + " mm" : "-"; if (d.snowfall_sum[i] != null && d.snowfall_sum[i] > 0) precipDisplay = d.snowfall_sum[i] + " cm (neve)"; htmlRows += ` ${d.time[i]} ${getTextCondition(d.weather_code[i])} ${d.temperature_2m_max[i] ?? '-'}°C ${d.apparent_temperature_max[i] ?? '-'}°C ${d.temperature_2m_min[i] ?? '-'}°C ${d.relative_humidity_2m_mean[i] ?? '-'}% ${d.pressure_msl_mean[i] ?? '-'} hPa ${d.wind_speed_10m_max[i] ?? '-'} km/h (${d.wind_direction_10m_dominant[i] ?? '-'}°) ${precipDisplay} `; } tbody.innerHTML = htmlRows; } catch (error) { console.error(error); tbody.innerHTML = "Errore nel caricamento. Riprova."; } }
